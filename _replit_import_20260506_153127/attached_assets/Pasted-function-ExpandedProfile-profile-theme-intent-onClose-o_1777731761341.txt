  return (
    <motion.div
      className="fixed inset-0 z-50 flex justify-center bg-black text-white"
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0 }}
    >
      <motion.div
        className="relative h-screen w-full max-w-[430px] overflow-y-auto bg-[#050505] pb-32 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden"

        exit={{ y: "100%" }}
        transition={{ type: "spring", stiffness: 220, damping: 28 }}
      >
        <div className="relative h-[58vh] min-h-[470px] overflow-hidden">
          <img
            src={profile.image}
            alt={profile.name}
            className="h-full w-full object-cover"
          />

          <div className="absolute inset-0 bg-gradient-to-t from-[#050505] via-black/30 to-black/20" />

          <div className="absolute left-4 right-4 top-5 flex items-center justify-between">

              className="grid h-12 w-12 place-items-center rounded-full border border-white/10 bg-black/50 backdrop-blur-xl"
            >
              <X className="h-6 w-6" />
            </button>

            <button className="grid h-12 w-12 place-items-center rounded-full border border-white/10 bg-black/50 backdrop-blur-xl">
              <SlidersHorizontal className="h-5 w-5" />

          </div>

          <div className="absolute bottom-0 left-0 right-0 p-5">
            <div className="flex items-end justify-between gap-4">
              <div>
                <div className="flex items-center gap-2">
                  <h1 className="text-5xl font-black tracking-tight">
                    {profile.name}, {profile.age}
                  </h1>

                  {profile.verified && (
                    <ShieldCheck className="h-7 w-7 fill-pink-500 text-white" />
                  )}
                </div>

                  <MapPin className="h-4 w-4" />

                    Online
                  </span>
                </div>
              </div>

              <div className="rounded-full border border-pink-400/30 bg-pink-500/15 px-4 py-2 text-sm font-black text-pink-200 backdrop-blur-xl">
                {profile.matchScore}% Match

              <span
                className={`rounded-full bg-gradient-to-r ${theme.gradient} px-4 py-1.5 text-xs font-black capitalize`}
              >
                {profile.intent}
              </span>

              <span className="rounded-full border border-white/10 bg-white/15 px-4 py-1.5 text-xs font-bold backdrop-blur-xl">
                {profile.subGenre}
              </span>

function InfoPill({ label, value }) {
  return (
    <div className="rounded-2xl border border-white/10 bg-white/[0.06] p-3 backdrop-blur-xl">
      <p className="text-[11px] font-bold uppercase tracking-wide text-zinc-500">
        {label}
      </p>
      <p className="mt-1 text-sm font-black text-white">{value}</p>
    </div>
  );
}

function MiniPhoto({ image }) {
  return (
    <div className="h-28 overflow-hidden rounded-2xl border border-white/10 bg-white/10">
      <img src={image} alt="" className="h-full w-full object-cover" />
    </div>
  );
}

function BigAction({ icon: Icon, label, main, theme, onClick }) {
  return (
    <button
      onClick={onClick}
      className="flex flex-col items-center gap-2 text-xs font-bold text-zinc-200 active:scale-95 transition"
    >
      <div
        className={`grid h-16 w-16 place-items-center rounded-full border backdrop-blur-xl ${
          main
            ? `border-white/20 bg-gradient-to-r ${theme.gradient} ${theme.glow}`
            : "border-white/15 bg-black/50"
        }`}
      >
        <Icon className={`h-8 w-8 ${main ? "text-white fill-white/20" : "text-zinc-200"}`} />
      </div>
      {label}
    </button>
  );
}
