import React, { useMemo, useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { Heart, Sparkles, X, MessageCircle } from "lucide-react";

const currentUser = {
  id: "user_1",
  name: "You",
};

const initialProfiles = [
  {
    id: "user_2",
    name: "Maya",
    age: 24,
    location: "Miami, FL",
    intent: "dating",
    likedCurrentUser: true,
    image: "https://images.unsplash.com/photo-1517841905240-472988babdf9?auto=format&fit=crop&w=900&q=80",
  },
  {
    id: "user_3",
    name: "Sofia",
    age: 25,
    location: "Brickell, Miami",
    intent: "dating",
    likedCurrentUser: false,
    image: "https://images.unsplash.com/photo-1524504388940-b1c1722653e1?auto=format&fit=crop&w=900&q=80",
  },
];

export default function DatingDiscoverSystem() {
  const [profiles, setProfiles] = useState(initialProfiles);
  const [likes, setLikes] = useState([]);
  const [passes, setPasses] = useState([]);
  const [matches, setMatches] = useState([]);
  const [chats, setChats] = useState([]);
  const [matchModal, setMatchModal] = useState(null);
  const [index, setIndex] = useState(0);

  const visibleProfiles = useMemo(() => {
    return profiles.filter((profile) => {
      const alreadyLiked = likes.some(
        (like) => like.fromUserId === currentUser.id && like.toUserId === profile.id
      );

      const alreadyPassed = passes.some(
        (pass) => pass.fromUserId === currentUser.id && pass.toUserId === profile.id
      );

      return !alreadyLiked && !alreadyPassed;
    });
  }, [profiles, likes, passes]);

  const profile = visibleProfiles[index % Math.max(visibleProfiles.length, 1)];

  function nextProfile() {
    setIndex((prev) => prev + 1);
  }

  function handlePass(profile) {
    const newPass = {
      id: crypto.randomUUID(),
      fromUserId: currentUser.id,
      toUserId: profile.id,
      createdAt: new Date().toISOString(),
    };

    setPasses((prev) => [...prev, newPass]);
    nextProfile();
  }

  function handleLike(profile, type = "vibe") {
    const newLike = {
      id: crypto.randomUUID(),
      fromUserId: currentUser.id,
      toUserId: profile.id,
      type,
      createdAt: new Date().toISOString(),
    };

    setLikes((prev) => [...prev, newLike]);

    const isMutual =
      profile.likedCurrentUser ||
      likes.some(
        (like) =>
          like.fromUserId === profile.id &&
          like.toUserId === currentUser.id
      );

    if (isMutual) {
      createMatch(profile);
    }

    nextProfile();
  }

  function createMatch(profile) {
    const chatId = crypto.randomUUID();

    const newChat = {
      id: chatId,
      participantIds: [currentUser.id, profile.id],
      type: "dating_match",
      messages: [
        {
          id: crypto.randomUUID(),
          senderId: "system",
          text: `You matched with ${profile.name}. Say something fun 👀`,
          createdAt: new Date().toISOString(),
        },
      ],
    };

    const newMatch = {
      id: crypto.randomUUID(),
      userIds: [currentUser.id, profile.id],
      profile,
      chatId,
      createdAt: new Date().toISOString(),
    };

    setChats((prev) => [...prev, newChat]);
    setMatches((prev) => [...prev, newMatch]);
    setMatchModal({ profile, chatId });
  }

  return (
    <div className="min-h-screen bg-black text-white">
      <div className="mx-auto max-w-[430px] px-4 py-6">
        <h1 className="text-3xl font-black">Dating Discover</h1>

        {profile ? (
          <div className="relative mt-6 h-[620px] overflow-hidden rounded-[34px] border border-pink-400/40 bg-zinc-950 shadow-[0_0_50px_rgba(236,72,153,0.3)]">
            <img
              src={profile.image}
              alt={profile.name}
              className="absolute inset-0 h-full w-full object-cover"
            />

            <div className="absolute inset-0 bg-gradient-to-t from-black via-black/35 to-transparent" />

            <div className="absolute bottom-0 left-0 right-0 p-6">
              <h2 className="text-4xl font-black">
                {profile.name}, {profile.age}
              </h2>

              <p className="mt-2 text-sm text-zinc-300">{profile.location}</p>

              <div className="mt-6 grid grid-cols-3 gap-3">
                <button
                  onClick={() => handlePass(profile)}
                  className="rounded-2xl border border-rose-400/30 bg-rose-500/10 py-4 text-rose-300"
                >
                  <X className="mx-auto h-7 w-7" />
                  <p className="mt-1 text-xs font-black">PASS</p>
                </button>

                <button
                  onClick={() => handleLike(profile, "spark")}
                  className="rounded-2xl border border-purple-400/30 bg-purple-500/10 py-4 text-purple-300"
                >
                  <Sparkles className="mx-auto h-7 w-7" />
                  <p className="mt-1 text-xs font-black">SPARK</p>
                </button>

                <button
                  onClick={() => handleLike(profile, "vibe")}
                  className="rounded-2xl border border-pink-400/30 bg-pink-500/10 py-4 text-pink-300"
                >
                  <Heart className="mx-auto h-7 w-7 fill-pink-300" />
                  <p className="mt-1 text-xs font-black">VIBE</p>
                </button>
              </div>
            </div>
          </div>
        ) : (
          <div className="mt-10 rounded-3xl border border-white/10 bg-white/[0.06] p-8 text-center">
            <p className="text-2xl font-black">No more profiles</p>
            <p className="mt-2 text-zinc-400">Check back soon.</p>
          </div>
        )}

        <ConnectTabPreview matches={matches} chats={chats} />
      </div>

      <AnimatePresence>
        {matchModal && (
          <MatchModal
            profile={matchModal.profile}
            onClose={() => setMatchModal(null)}
          />
        )}
      </AnimatePresence>
    </div>
  );
}

function MatchModal({ profile, onClose }) {
  return (
    <motion.div
      className="fixed inset-0 z-50 grid place-items-center bg-black/80 px-6 backdrop-blur-xl"
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0 }}
    >
      <motion.div
        initial={{ scale: 0.85, y: 30 }}
        animate={{ scale: 1, y: 0 }}
        exit={{ scale: 0.9, y: 20 }}
        className="w-full max-w-[360px] rounded-[36px] border border-pink-400/40 bg-[#12000b] p-6 text-center shadow-[0_0_80px_rgba(236,72,153,0.45)]"
      >
        <div className="mx-auto h-28 w-28 overflow-hidden rounded-full border-4 border-pink-400">
          <img
            src={profile.image}
            alt={profile.name}
            className="h-full w-full object-cover"
          />
        </div>

        <h2 className="mt-5 text-4xl font-black text-white">It’s a Vibe</h2>

        <p className="mt-2 text-sm text-zinc-300">
          You and {profile.name} both liked each other.
        </p>

        <button className="mt-6 w-full rounded-full bg-gradient-to-r from-pink-500 to-fuchsia-500 py-3 text-sm font-black text-white">
          Message Now
        </button>

        <button
          onClick={onClose}
          className="mt-3 w-full rounded-full border border-white/10 bg-white/5 py-3 text-sm font-bold text-zinc-300"
        >
          Keep Discovering
        </button>
      </motion.div>
    </motion.div>
  );
}

function ConnectTabPreview({ matches }) {
  return (
    <section className="mt-8">
      <h2 className="text-xl font-black">Connect Tab Matches</h2>

      {matches.length === 0 ? (
        <p className="mt-3 text-sm text-zinc-500">
          Your dating matches will appear here.
        </p>
      ) : (
        <div className="mt-4 space-y-3">
          {matches.map((match) => (
            <div
              key={match.id}
              className="flex items-center gap-3 rounded-3xl border border-white/10 bg-white/[0.06] p-3"
            >
              <img
                src={match.profile.image}
                alt={match.profile.name}
                className="h-14 w-14 rounded-full object-cover"
              />

              <div className="flex-1">
                <p className="font-black">{match.profile.name}</p>
                <p className="text-xs text-zinc-400">New match · Say hi</p>
              </div>

              <button className="grid h-10 w-10 place-items-center rounded-full bg-pink-500">
                <MessageCircle className="h-5 w-5" />
              </button>
            </div>
