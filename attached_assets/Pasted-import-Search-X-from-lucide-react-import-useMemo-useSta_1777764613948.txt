import { Search, X } from "lucide-react";
import { useMemo, useState } from "react";

function NetworkingSearch({ people, opportunities, groups }) {
  const [query, setQuery] = useState("");
  const [filter, setFilter] = useState("All");

  const filters = ["All", "Jobs", "Internships", "People", "Groups", "Events", "Open Doors"];

  const results = useMemo(() => {
    const q = query.toLowerCase().trim();

    const peopleResults = people.map((p) => ({
      type: "People",
      title: p.name,
      subtitle: `${p.role} • ${p.org}`,
      tags: p.skills,
      data: p,
    }));

    const opportunityResults = opportunities.map((o) => ({
      type: o.type || "Jobs",
      title: o.title,
      subtitle: `${o.company} • ${o.location}`,
      tags: o.tags || [o.type],
      data: o,
    }));

    const groupResults = groups.map((g) => ({
      type: "Groups",
      title: g.title,
      subtitle: `${g.members} members • ${g.online}`,
      tags: ["Group", "Networking"],
      data: g,
    }));

    return [...peopleResults, ...opportunityResults, ...groupResults].filter((item) => {
      const matchesFilter = filter === "All" || item.type === filter;

      const searchable = [
        item.type,
        item.title,
        item.subtitle,
        ...(item.tags || []),
      ]
        .join(" ")
        .toLowerCase();

      const matchesQuery = q === "" || searchable.includes(q);

      return matchesFilter && matchesQuery;
    });
  }, [query, filter, people, opportunities, groups]);

  return (
    <section className="mt-5">
      <div className="relative">
        <Search className="absolute left-4 top-1/2 h-5 w-5 -translate-y-1/2 text-pink-300" />

        <input
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Search jobs, people, groups, open doors..."
          className="w-full rounded-full border border-pink-400/25 bg-black/55 py-3 pl-12 pr-12 text-sm font-semibold text-white outline-none backdrop-blur-xl placeholder:text-zinc-500 focus:border-pink-400 focus:shadow-[0_0_30px_rgba(236,72,153,0.28)]"
        />

        {query && (
          <button
            onClick={() => setQuery("")}
            className="absolute right-4 top-1/2 -translate-y-1/2 text-zinc-400"
          >
            <X className="h-5 w-5" />
          </button>
        )}
      </div>

      <div className="mt-3 flex gap-2 overflow-x-auto pb-1 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden">
        {filters.map((item) => (
          <button
            key={item}
            onClick={() => setFilter(item)}
            className={`shrink-0 rounded-full px-4 py-2 text-xs font-black transition ${
              filter === item
                ? "bg-gradient-to-r from-pink-500 to-fuchsia-500 text-white shadow-[0_0_22px_rgba(236,72,153,0.45)]"
                : "border border-white/10 bg-white/[0.06] text-zinc-400"
            }`}
          >
            {item}
          </button>
        ))}
      </div>

      {query && (
        <div className="mt-4 space-y-3">
          {results.length > 0 ? (
            results.map((item, index) => (
              <div
                key={`${item.type}-${item.title}-${index}`}
                className="rounded-3xl border border-white/10 bg-white/[0.06] p-4 backdrop-blur-xl"
              >
                <div className="flex items-start justify-between gap-3">
                  <div>
                    <span className="rounded-full bg-pink-500/15 px-3 py-1 text-[10px] font-black text-pink-300">
                      {item.type}
                    </span>

                    <h3 className="mt-3 text-sm font-black text-white">
                      {item.title}
                    </h3>

                    <p className="mt-1 text-xs text-zinc-400">
                      {item.subtitle}
                    </p>

                    <div className="mt-3 flex flex-wrap gap-2">
                      {item.tags?.slice(0, 3).map((tag) => (
                        <span
                          key={tag}
                          className="rounded-full border border-white/10 bg-black/35 px-2.5 py-1 text-[10px] font-semibold text-zinc-300"
                        >
                          {tag}
                        </span>
                      ))}
                    </div>
                  </div>

                  <button className="rounded-full bg-pink-500 px-4 py-2 text-xs font-black text-white">
                    Open
                  </button>
                </div>
              </div>
            ))
          ) : (
            <div className="rounded-3xl border border-pink-400/20 bg-pink-500/10 p-6 text-center">
              <p className="text-lg font-black text-white">Nothing found yet</p>
              <p className="mt-2 text-sm text-zinc-400">
                Post an open door and help someone connect.
              </p>
            </div>
          )}
        </div>
      )}
    </section>
  );
}
