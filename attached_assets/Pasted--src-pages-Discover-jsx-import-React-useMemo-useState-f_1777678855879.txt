export default function Discover() {
  const allowedTabs =
    currentUserIntent === "all"
      ? intentTabs
      : intentTabs.filter((tab) => tab.id === currentUserIntent);

  const [activeIntent, setActiveIntent] = useState(allowedTabs[0].id);
  const [activeSubTab, setActiveSubTab] = useState("For You");
  const [cardIndex, setCardIndex] = useState(0);

  const activeTheme = intentTabs.find((tab) => tab.id === activeIntent);

  const filteredProfiles = useMemo(() => {
    return profiles.filter((profile) => {
      const userCanSee =
        currentUserIntent === "all" ||
        profile.intent === currentUserIntent ||
        profile.intent === "all";

      const matchesMainTab =
        profile.intent === activeIntent || profile.intent === "all";

      const matchesSubTab =
        activeSubTab === "For You" || profile.subGenre === activeSubTab;

      return userCanSee && matchesMainTab && matchesSubTab;
    });
  }, [activeIntent, activeSubTab]);

  const currentProfile = filteredProfiles[cardIndex % Math.max(filteredProfiles.length, 1)];

  const changeIntent = (intent) => {
    setActiveIntent(intent);
    setActiveSubTab("For You");
    setCardIndex(0);
  };

  const nextCard = () => {
    setCardIndex((prev) => prev + 1);
  };

  return (
    <div className="min-h-screen w-full bg-black text-white flex justify-center">
      <div className="relative min-h-screen w-full max-w-[430px] overflow-hidden bg-[#050505] pb-24">
        <div className="absolute inset-0 bg-[radial-gradient(circle_at_top_left,rgba(236,72,153,0.16),transparent_35%),radial-gradient(circle_at_top_right,rgba(16,185,129,0.12),transparent_35%)]" />

        <main className="relative z-10 px-4 pt-5">
          <header className="flex items-start justify-between">
            <div>
              <h1 className="text-[34px] font-black tracking-tight">Discover</h1>
              <div className="mt-1 flex items-center gap-2 text-sm text-zinc-400">
                <Flame className="h-4 w-4 text-pink-500 fill-pink-500" />
                <span>9 people waiting</span>
                <span className="h-2 w-2 rounded-full bg-emerald-400 shadow-[0_0_12px_rgba(52,211,153,0.9)]" />
              </div>
            </div>

            <button className="mt-2 grid h-12 w-12 place-items-center rounded-full border border-white/10 bg-white/5 backdrop-blur-xl">
              <SlidersHorizontal className="h-5 w-5 text-zinc-200" />
            </button>
          </header>

          <section className="mt-5 rounded-full border border-white/10 bg-zinc-950/80 p-1 shadow-2xl backdrop-blur-xl">
            <div className="grid grid-cols-3 gap-1">
              {intentTabs.map((tab) => {
                const Icon = tab.icon;
                const isAllowed = allowedTabs.some((allowed) => allowed.id === tab.id);
                const isActive = activeIntent === tab.id;

                return (
                  <button
                    key={tab.id}
                    disabled={!isAllowed}
                    onClick={() => isAllowed && changeIntent(tab.id)}
                    className={`relative flex items-center justify-center gap-2 rounded-full px-3 py-3 text-xs font-bold transition ${
                      isActive
                        ? "text-white"
                        : isAllowed
                        ? "text-zinc-400"
                        : "text-zinc-700 opacity-40"
                    }`}
                  >
                    {isActive && (
                      <motion.div
                        layoutId="activeIntent"
                        className={`absolute inset-0 rounded-full bg-gradient-to-r ${tab.accent}`}
                        transition={{ type: "spring", stiffness: 400, damping: 32 }}
                      />
                    )}
                    <span className="relative z-10 flex items-center gap-2">
                      <Icon className="h-4 w-4" />
                      {tab.label}
                    </span>
                  </button>
                );
              })}
            </div>
          </section>

          <section className="mt-3 flex gap-2 overflow-x-auto pb-1 scrollbar-hide">
            {subTabs[activeIntent].map((tab) => {
              const active = activeSubTab === tab;

              return (
                <button
                  key={tab}
                  onClick={() => {
                    setActiveSubTab(tab);
                    setCardIndex(0);
                  }}
                  className={`shrink-0 rounded-full border px-4 py-2 text-xs font-bold transition ${
                    active
                      ? `border-transparent bg-gradient-to-r ${activeTheme.accent} text-white`
                      : "border-white/10 bg-white/5 text-zinc-400"
                  }`}
                >
                  {tab}
                </button>
              );
            })}
          </section>

          <div className="mt-3 rounded-full border border-pink-400/20 bg-pink-500/10 px-4 py-2 text-center text-xs font-semibold text-pink-200">
            ✨ Showing mock profiles while your real feed is empty.
          </div>

          <section className="relative mt-4 h-[590px]">
            <AnimatePresence mode="wait">
              {currentProfile ? (
                <motion.article
                  key={`${currentProfile.id}-${cardIndex}-${activeIntent}-${activeSubTab}`}
                  drag="y"
                  dragConstraints={{ top: 0, bottom: 0 }}
                  onDragEnd={(event, info) => {
                    if (Math.abs(info.offset.y) > 90) nextCard();
                  }}
                  initial={{ opacity: 0, y: 80, scale: 0.96 }}
                  animate={{ opacity: 1, y: 0, scale: 1 }}
                  exit={{ opacity: 0, y: -120, scale: 0.94 }}
                  transition={{ type: "spring", stiffness: 220, damping: 26 }}
                  className="absolute inset-0 overflow-hidden rounded-[28px] border border-white/10 bg-zinc-900 shadow-[0_30px_80px_rgba(0,0,0,0.75)]"
                >
                  <img
                    src={currentProfile.image}
                    alt={currentProfile.name}
                    className="h-full w-full object-cover"
                  />

                  <div className="absolute inset-0 bg-gradient-to-t from-black via-black/25 to-black/10" />

                  <div className="absolute left-5 right-5 top-4 flex items-center justify-between">
                    <div className="rounded-full bg-black/45 px-4 py-2 text-xs font-bold backdrop-blur-xl">
                      <span
                        className={`mr-2 inline-block h-2 w-2 rounded-full ${
                          currentProfile.online ? "bg-emerald-400" : "bg-zinc-500"
                        }`}
                      />
                      {currentProfile.online ? "Online" : "Offline"}
                    </div>

                    <div className="rounded-full bg-black/45 px-4 py-2 text-xs font-bold backdrop-blur-xl">
                      {currentProfile.matchScore}% Match
                    </div>
                  </div>

                  <div className="absolute bottom-0 left-0 right-0 p-5">
                    <div className="flex items-end justify-between gap-4">
                      <div>
                        <h2 className="text-4xl font-black tracking-tight">
                          {currentProfile.name}, {currentProfile.age}
                        </h2>

                        <div className="mt-2 flex items-center gap-1 text-sm text-zinc-200">
                          <MapPin className="h-4 w-4" />
                          {currentProfile.location}
                        </div>
                      </div>

                      <button
                        onClick={nextCard}
                        className={`grid h-13 w-13 place-items-center rounded-full bg-gradient-to-r ${activeTheme.accent} shadow-[0_0_30px_rgba(236,72,153,0.55)]`}
                      >
                        <CheckCircle className="h-7 w-7 fill-white text-white" />
                      </button>
                    </div>

                    <div className="mt-3 flex flex-wrap gap-2">
                      <span className={`rounded-full bg-gradient-to-r ${activeTheme.accent} px-3 py-1 text-xs font-black`}>
                        {currentProfile.intent}
                      </span>
                      <span className="rounded-full bg-white/15 px-3 py-1 text-xs font-bold backdrop-blur-xl">
                        {currentProfile.subGenre}
                      </span>
                    </div>

                    <p className="mt-3 line-clamp-2 text-sm font-medium leading-5 text-zinc-100">
                      {currentProfile.bio}
                    </p>

                    <div className="mt-3 flex flex-wrap gap-2">
                      {currentProfile.interests.map((interest) => (
                        <span
                          key={interest}
                          className="rounded-full bg-white/15 px-3 py-1.5 text-xs font-bold text-white backdrop-blur-xl"
                        >
                          {interest}
                        </span>
                      ))}
                    </div>

                    <div className="mt-5 grid grid-cols-3 gap-3">
                      {activeIntent === "dating" && (
                        <>
                          <ActionButton icon={X} label="Pass" onClick={nextCard} />
                          <ActionButton icon={Heart} label="Like" hot onClick={nextCard} />
                          <ActionButton icon={MessageCircle} label="Message" onClick={nextCard} />
                        </>
                      )}

                      {activeIntent === "friends" && (
                        <>
                          <ActionButton icon={X} label="Skip" onClick={nextCard} />
                          <ActionButton icon={UserPlus} label="Add Friend" hot onClick={nextCard} />
                          <ActionButton icon={CalendarPlus} label="Invite" onClick={nextCard} />
                        </>
                      )}

                      {activeIntent === "networking" && (
                        <>
                          <ActionButton icon={Handshake} label="Connect" hot onClick={nextCard} />
                          <ActionButton icon={Bookmark} label="Save" onClick={nextCard} />
                          <ActionButton icon={MessageCircle} label="Message" onClick={nextCard} />
                        </>
                      )}
                    </div>
                  </div>
                </motion.article>
              ) : (
                <div className="grid h-full place-items-center rounded-[28px] border border-white/10 bg-white/5 p-8 text-center">
                  <div>
                    <p className="text-xl font-black">No profiles here yet</p>
                    <p className="mt-2 text-sm text-zinc-400">
                      Try another sub-tab or intent category.
                    </p>
                  </div>
                </div>
              )}
            </AnimatePresence>
          </section>
        </main>

        <BottomNav />
      </div>
    </div>
  );
}

function ActionButton({ icon: Icon, label, hot, onClick }) {
  return (
    <button
      onClick={onClick}
      className={`flex flex-col items-center justify-center gap-1 rounded-2xl border py-3 text-xs font-black backdrop-blur-xl transition active:scale-95 ${
        hot
          ? "border-pink-400/40 bg-pink-500/20 text-white"
          : "border-white/10 bg-white/10 text-zinc-200"
      }`}
    >
      <Icon className="h-5 w-5" />
      {label}
    </button>
  );
}

function BottomNav() {
  const items = [
    { label: "Discover", icon: Flame, active: true },
    { label: "Connect", icon: Compass },
    { label: "Events", icon: CalendarDays },
    { label: "Map", icon: Map },
    { label: "Profile", icon: UserCircle },
  ];

  return (
    <nav className="absolute bottom-0 left-0 right-0 z-20 border-t border-white/10 bg-black/85 px-3 py-2 backdrop-blur-2xl">
      <div className="grid grid-cols-5">
        {items.map((item) => {
          const Icon = item.icon;

          return (
            <button
              key={item.label}
              className={`flex flex-col items-center gap-1 py-2 text-[11px] font-bold ${
                item.active ? "text-pink-500" : "text-zinc-500"
              }`}
            >
              <Icon className={`h-5 w-5 ${item.active ? "fill-pink-500/20" : ""}`} />
              {item.label}
            </button>
          );
        })}
      </div>
    </nav>
  );
}
