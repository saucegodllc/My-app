function NetworkingSearch({ people, opportunities, groups }) {
  const [query, setQuery] = useState("");
  const [filter, setFilter] = useState("All");

  const filters = ["All", "Jobs", "Internships", "People", "Groups", "Events"];

  const results = useMemo(() => {
    const q = query.toLowerCase().trim();

    const peopleResults = people.map((p) => ({
      type: "People",
      title: p.name,
      subtitle: `${p.role} • ${p.org}`,
      tags: p.skills || [],
      action: "View",
    }));

    const opportunityResults = opportunities.map((o) => ({
      type: o.type || "Jobs",
      title: o.title,
      subtitle: `${o.company} • ${o.location}`,
      tags: o.tags || [o.type],
      action: "Apply",
    }));

    const groupResults = groups.map((g) => ({
      type: "Groups",
      title: g.title,
      subtitle: `${g.members} members • ${g.online}`,
      tags: ["Group", "Networking"],
      action: "Join",
    }));

    const allResults = [...opportunityResults, ...peopleResults, ...groupResults];

    return allResults.filter((item) => {
      const matchesFilter =
        filter === "All" ||
        item.type === filter ||
        item.tags?.includes(filter);

      const searchable = [
        item.type,
        item.title,
        item.subtitle,
        ...(item.tags || []),
      ]
        .join(" ")
        .toLowerCase();

      const matchesQuery = q === "" || searchable.includes(q);

      return matchesFilter && matchesQuery;
    });
  }, [query, filter, people, opportunities, groups]);

  return (
    <section className="mt-4">
      <div className="rounded-[28px] border border-pink-400/25 bg-black/55 p-3 shadow-[0_0_35px_rgba(236,72,153,0.18)] backdrop-blur-2xl">
        <div className="relative">
          <Search className="absolute left-4 top-1/2 h-5 w-5 -translate-y-1/2 text-pink-300" />

          <input
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            placeholder="Search jobs, internships, people, groups..."
            className="h-12 w-full rounded-full border border-white/10 bg-white/[0.06] pl-12 pr-11 text-sm font-semibold text-white outline-none placeholder:text-zinc-500 focus:border-pink-400/60 focus:shadow-[0_0_25px_rgba(236,72,153,0.25)]"
          />

          {query && (
            <button
              onClick={() => setQuery("")}
              className="absolute right-4 top-1/2 -translate-y-1/2 text-zinc-400"
            >
              <X className="h-5 w-5" />
            </button>
          )}
        </div>

        <div className="mt-3 flex gap-2 overflow-x-auto pb-1 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden">
          {filters.map((item) => (
            <button
              key={item}
              onClick={() => setFilter(item)}
              className={`shrink-0 rounded-full px-4 py-2 text-xs font-black transition active:scale-95 ${
                filter === item
                  ? "bg-gradient-to-r from-pink-500 to-fuchsia-500 text-white shadow-[0_0_22px_rgba(236,72,153,0.45)]"
                  : "border border-white/10 bg-white/[0.06] text-zinc-400"
              }`}
            >
              {item}
            </button>
          ))}
        </div>
      </div>

      {(query || filter !== "All") && (
        <div className="mt-4">
          <div className="mb-3 flex items-center justify-between px-1">
            <p className="text-sm font-black text-white">
              {results.length} results
            </p>

            <p className="text-xs font-semibold text-pink-300">
              Open doors nearby
            </p>
          </div>

          <div className="space-y-3">
            {results.length > 0 ? (
              results.map((item, index) => (
                <SearchResultCard key={`${item.title}-${index}`} item={item} />
              ))
            ) : (
              <div className="rounded-[28px] border border-pink-400/20 bg-pink-500/10 p-6 text-center">
                <p className="text-lg font-black text-white">No results yet</p>
                <p className="mt-2 text-sm text-zinc-400">
                  Post an open door and help someone connect.
                </p>
              </div>
            )}
          </div>
        </div>
      )}
    </section>
  );
}

function SearchResultCard({ item }) {
  const typeStyles = {
    People: "bg-pink-500/15 text-pink-300 border-pink-400/20",
    Jobs: "bg-emerald-500/15 text-emerald-300 border-emerald-400/20",
    Internships: "bg-blue-500/15 text-blue-300 border-blue-400/20",
    Groups: "bg-purple-500/15 text-purple-300 border-purple-400/20",
    Events: "bg-orange-500/15 text-orange-300 border-orange-400/20",
  };

  return (
    <article className="rounded-[26px] border border-white/10 bg-white/[0.06] p-4 backdrop-blur-xl transition active:scale-[0.98]">
      <div className="flex items-start justify-between gap-3">
        <div className="min-w-0">
          <span
            className={`inline-flex rounded-full border px-3 py-1 text-[10px] font-black ${
              typeStyles[item.type] || typeStyles.Jobs
            }`}
          >
            {item.type}
          </span>

          <h3 className="mt-3 truncate text-base font-black text-white">
            {item.title}
          </h3>

          <p className="mt-1 line-clamp-1 text-xs font-semibold text-zinc-400">
            {item.subtitle}
          </p>
        </div>

        <button className="shrink-0 rounded-full bg-gradient-to-r from-pink-500 to-fuchsia-500 px-4 py-2 text-xs font-black text-white shadow-[0_0_22px_rgba(236,72,153,0.35)]">
          {item.action}
        </button>
      </div>

      <div className="mt-3 flex flex-wrap gap-2">
        {item.tags?.slice(0, 4).map((tag) => (
          <span
            key={tag}
            className="rounded-full border border-white/10 bg-black/35 px-2.5 py-1 text-[10px] font-semibold text-zinc-300"
          >
            {tag}
          </span>
        ))}
      </div>
    </article>
  );
}
