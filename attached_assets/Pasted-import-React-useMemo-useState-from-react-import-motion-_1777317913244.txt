import React, { useMemo, useState } from "react";
import { motion, AnimatePresence, useMotionValue, useTransform, animate } from "framer-motion";
import { Flame, SlidersHorizontal, X, Star, Heart, MapPin, BadgeCheck, Info, Zap } from "lucide-react";

const INTENTS = {
  Dating: ["All", "Serious", "Casual", "Double Dates", "Active Tonight", "New in Town"],
  Friends: ["All", "Going Out", "Gym", "Study", "Travel", "Foodies", "Nightlife"],
  Networking: ["All", "Entrepreneurs", "Creators", "Students", "Investors", "Jobs", "Mentors"],
};

const DEFAULT_USERS = [];

// This component is built for real users, not fake model profiles.
// Connect your backend data by passing users into <ViralDiscoverTab users={yourUsers} />.
// Expected user shape:
// {
//   id, name, age, location, intent, subIntent, verified, online,
//   compatibility, vibe, bio, interests, photo
// }

function getInitials(name = "User") {
  return name
    .split(" ")
    .map((part) => part[0])
    .join("")
    .slice(0, 2)
    .toUpperCase();
}

function IntentTabs({ active, setActive }) {
  return (
    <div className="mx-5 mt-4 rounded-full bg-white/5 p-1 border border-white/10 flex">
      {Object.keys(INTENTS).map((intent) => (
        <button
          key={intent}
          onClick={() => setActive(intent)}
          className={`relative flex-1 rounded-full py-3 text-sm font-semibold transition ${active === intent ? "text-white" : "text-zinc-400"}`}
        >
          {active === intent && (
            <motion.div
              layoutId="intent-bg"
              className="absolute inset-0 rounded-full bg-gradient-to-r from-pink-500 to-fuchsia-700 shadow-[0_0_24px_rgba(236,72,153,.45)]"
              transition={{ type: "spring", stiffness: 380, damping: 32 }}
            />
          )}
          <span className="relative z-10">{intent}</span>
        </button>
      ))}
    </div>
  );
}

function SubIntentChips({ intent, active, setActive }) {
  return (
    <div className="mt-4 flex gap-2 overflow-x-auto px-5 pb-1 no-scrollbar">
      {INTENTS[intent].map((chip) => (
        <button
          key={chip}
          onClick={() => setActive(chip)}
          className={`whitespace-nowrap rounded-full border px-4 py-2 text-xs font-bold transition ${
            active === chip
              ? "border-pink-400 bg-pink-500 text-white shadow-[0_0_18px_rgba(236,72,153,.45)]"
              : "border-white/10 bg-white/5 text-zinc-300"
          }`}
        >
          {chip}
        </button>
      ))}
    </div>
  );
}

function SwipeCard({ user, isTop, onSwipe, onOpen }) {
  const x = useMotionValue(0);
  const y = useMotionValue(0);
  const rotate = useTransform(x, [-240, 0, 240], [-14, 0, 14]);
  const likeOpacity = useTransform(x, [40, 150], [0, 1]);
  const passOpacity = useTransform(x, [-150, -40], [1, 0]);
  const superOpacity = useTransform(y, [-170, -60], [1, 0]);

  const finishSwipe = async (direction) => {
    if (direction === "like") await animate(x, 760, { duration: 0.24, ease: "easeOut" });
    if (direction === "pass") await animate(x, -760, { duration: 0.24, ease: "easeOut" });
    if (direction === "super") await animate(y, -920, { duration: 0.24, ease: "easeOut" });
    onSwipe(direction, user);
  };

  const handleDragEnd = () => {
    const currentX = x.get();
    const currentY = y.get();
    if (currentX > 130) return finishSwipe("like");
    if (currentX < -130) return finishSwipe("pass");
    if (currentY < -130) return finishSwipe("super");
    animate(x, 0, { type: "spring", stiffness: 430, damping: 30 });
    animate(y, 0, { type: "spring", stiffness: 430, damping: 30 });
  };

  const hasPhoto = Boolean(user.photo);

  return (
    <motion.div
      className="absolute inset-0 overflow-hidden rounded-[30px] bg-zinc-900 shadow-2xl border border-white/10"
      style={{ x, y, rotate }}
      drag={isTop}
      dragConstraints={{ left: 0, right: 0, top: 0, bottom: 0 }}
      dragElastic={0.18}
      whileTap={isTop ? { scale: 0.985 } : undefined}
      initial={{ scale: isTop ? 1 : 0.94, y: isTop ? 0 : 18, opacity: isTop ? 1 : 0.65 }}
      animate={{ scale: isTop ? 1 : 0.94, y: isTop ? 0 : 18, opacity: isTop ? 1 : 0.65 }}
      transition={{ type: "spring", stiffness: 300, damping: 28 }}
      onDragEnd={handleDragEnd}
      onDoubleClick={() => isTop && finishSwipe("like")}
    >
      {hasPhoto ? (
        <img src={user.photo} alt={user.name} className="h-full w-full object-cover" />
      ) : (
        <div className="grid h-full w-full place-items-center bg-gradient-to-br from-zinc-900 via-fuchsia-950 to-black">
          <div className="grid h-32 w-32 place-items-center rounded-full border border-white/15 bg-white/10 text-5xl font-black text-white shadow-2xl backdrop-blur-xl">
            {getInitials(user.name)}
          </div>
        </div>
      )}
      <div className="absolute inset-0 bg-gradient-to-t from-black via-black/30 to-black/5" />

      <motion.div style={{ opacity: likeOpacity }} className="absolute left-6 top-8 rotate-[-12deg] rounded-2xl border-2 border-pink-400 px-5 py-2 text-3xl font-black text-pink-300">LIKE</motion.div>
      <motion.div style={{ opacity: passOpacity }} className="absolute right-6 top-8 rotate-[12deg] rounded-2xl border-2 border-red-400 px-5 py-2 text-3xl font-black text-red-300">PASS</motion.div>
      <motion.div style={{ opacity: superOpacity }} className="absolute left-1/2 top-20 -translate-x-1/2 rounded-2xl border-2 border-purple-400 px-5 py-2 text-3xl font-black text-purple-300">SUPER</motion.div>

      <div className="absolute left-4 top-4 flex items-center gap-2 rounded-full bg-black/45 px-3 py-2 text-xs font-bold text-white backdrop-blur-xl">
        <span className={`h-2 w-2 rounded-full ${user.online ? "bg-green-400" : "bg-zinc-500"}`} />
        {user.online ? "Online" : "Recently active"}
      </div>
      <div className="absolute right-4 top-4 rounded-full bg-black/45 px-3 py-2 text-xs font-bold text-white backdrop-blur-xl">
        💗 {user.compatibility}% Match
      </div>

      <button onClick={onOpen} className="absolute bottom-24 right-4 rounded-full bg-black/40 p-2 text-white backdrop-blur-xl border border-white/10">
        <Info size={18} />
      </button>

      <div className="absolute inset-x-0 bottom-0 p-5 pt-28">
        <div className="flex items-center gap-2">
          <h2 className="text-3xl font-black tracking-tight text-white">{user.name}, {user.age}</h2>
          {user.verified && <BadgeCheck className="h-6 w-6 fill-pink-500 text-white" />}
        </div>
        <div className="mt-1 flex items-center gap-1 text-sm text-zinc-200"><MapPin size={14} /> {user.location}</div>
        <div className="mt-3 flex gap-2 overflow-hidden">
          <span className="rounded-full bg-pink-500/25 px-3 py-1 text-xs font-bold text-pink-100">{user.intent}</span>
          <span className="rounded-full bg-purple-500/25 px-3 py-1 text-xs font-bold text-purple-100">{user.subIntent}</span>
        </div>
        <p className="mt-3 line-clamp-2 text-sm font-medium text-white">{user.vibe}</p>
        <p className="mt-1 line-clamp-2 text-sm text-zinc-300">{user.bio}</p>
        <div className="mt-4 flex gap-2 overflow-x-auto no-scrollbar">
          {user.interests.map((interest) => <span key={interest} className="whitespace-nowrap rounded-full bg-white/12 px-3 py-2 text-xs font-bold text-white backdrop-blur">{interest}</span>)}
        </div>
      </div>
    </motion.div>
  );
}

function ActionButtons({ onPass, onSuper, onLike }) {
  return (
    <div className="mx-auto mt-5 flex items-center justify-center gap-7">
      <button onClick={onPass} className="grid h-16 w-16 place-items-center rounded-full bg-white/8 border border-white/10 text-zinc-300 shadow-xl backdrop-blur transition hover:scale-105"><X size={32} /></button>
      <button onClick={onSuper} className="grid h-16 w-16 place-items-center rounded-full bg-purple-500/15 border border-purple-400/30 text-purple-300 shadow-[0_0_28px_rgba(168,85,247,.22)] backdrop-blur transition hover:scale-105"><Star size={32} fill="currentColor" /></button>
      <button onClick={onLike} className="grid h-16 w-16 place-items-center rounded-full bg-pink-500/20 border border-pink-400/30 text-pink-300 shadow-[0_0_34px_rgba(236,72,153,.3)] backdrop-blur transition hover:scale-105"><Heart size={32} fill="currentColor" /></button>
    </div>
  );
}

function MatchModal({ user, onClose }) {
  return (
    <motion.div className="fixed inset-0 z-50 grid place-items-center bg-black/85 p-6 backdrop-blur-xl" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}>
      <motion.div initial={{ scale: .85, y: 30 }} animate={{ scale: 1, y: 0 }} exit={{ scale: .9, opacity: 0 }} className="w-full max-w-sm rounded-[32px] border border-white/10 bg-zinc-950 p-6 text-center shadow-2xl">
        <div className="mx-auto mb-4 grid h-20 w-20 place-items-center rounded-full bg-gradient-to-r from-pink-500 to-purple-700 text-4xl">🎉</div>
        <h2 className="text-3xl font-black text-white">It’s a Match!</h2>
        <p className="mt-2 text-zinc-400">You and {user?.name} liked each other.</p>
        {user?.photo ? (
          <img src={user.photo} className="mx-auto mt-5 h-28 w-28 rounded-full object-cover ring-4 ring-pink-500" />
        ) : (
          <div className="mx-auto mt-5 grid h-28 w-28 place-items-center rounded-full bg-gradient-to-br from-pink-500 to-purple-800 text-3xl font-black text-white ring-4 ring-pink-500">
            {getInitials(user?.name)}
          </div>
        )}
        <button className="mt-6 w-full rounded-full bg-gradient-to-r from-pink-500 to-purple-700 py-4 font-bold text-white">Send a Message</button>
        <button onClick={onClose} className="mt-3 w-full rounded-full bg-white/8 py-4 font-bold text-zinc-200">Keep Discovering</button>
      </motion.div>
    </motion.div>
  );
}

export default function ViralDiscoverTab({ users = DEFAULT_USERS }) {
  const [intent, setIntent] = useState("Dating");
  const [subIntent, setSubIntent] = useState("All");
  const [dismissed, setDismissed] = useState([]);
  const [matchUser, setMatchUser] = useState(null);
  const [preview, setPreview] = useState(null);

  const filteredUsers = useMemo(() => {
    return users.filter(u => u.intent === intent && (subIntent === "All" || u.subIntent === subIntent) && !dismissed.includes(u.id));
  }, [intent, subIntent, dismissed]);

  const handleSwipe = (direction, user) => {
    setDismissed((ids) => [...ids, user.id]);
    if (direction === "like" && user.compatibility >= 90) setMatchUser(user);
  };

  const topUser = filteredUsers[0];

  return (
    <div className="min-h-screen bg-[#050505] text-white font-sans overflow-hidden">
      <div className="mx-auto min-h-screen max-w-md bg-[#050505] pb-24">
        <header className="flex items-center justify-between px-5 pt-8">
          <div>
            <h1 className="text-3xl font-black tracking-tight">Discover</h1>
            <div className="mt-1 flex items-center gap-1 text-xs text-zinc-400"><Flame size={14} className="text-pink-400" /> Active Now <span className="h-2 w-2 rounded-full bg-green-400" /></div>
          </div>
          <button className="rounded-full border border-white/10 bg-white/5 p-3 text-zinc-200"><SlidersHorizontal size={20} /></button>
        </header>

        <IntentTabs active={intent} setActive={(next) => { setIntent(next); setSubIntent("All"); setDismissed([]); }} />
        <SubIntentChips intent={intent} active={subIntent} setActive={(chip) => { setSubIntent(chip); setDismissed([]); }} />

        <div className="relative mx-5 mt-4 h-[58vh] min-h-[480px]">
          <AnimatePresence>
            {filteredUsers.slice(0, 2).reverse().map((user, index, arr) => {
              const isTop = index === arr.length - 1;
              return <SwipeCard key={user.id} user={user} isTop={isTop} onSwipe={handleSwipe} onOpen={() => setPreview(user)} />;
            })}
          </AnimatePresence>

          {filteredUsers.length === 0 && (
            <div className="grid h-full place-items-center rounded-[30px] border border-white/10 bg-white/[.03] p-8 text-center">
              <div>
                <div className="mx-auto grid h-20 w-20 place-items-center rounded-full bg-gradient-to-r from-pink-500 to-purple-700 text-4xl"><Zap /></div>
                <h2 className="mt-5 text-2xl font-black">You’re all caught up!</h2>
                <p className="mt-2 text-sm text-zinc-400">Try another vibe or expand your search.</p>
                <button onClick={() => { setSubIntent("All"); setDismissed([]); }} className="mt-5 rounded-full bg-gradient-to-r from-pink-500 to-purple-700 px-6 py-3 font-bold">Reset Discover</button>
              </div>
            </div>
          )}
        </div>

        {topUser && <ActionButtons onPass={() => handleSwipe("pass", topUser)} onSuper={() => handleSwipe("super", topUser)} onLike={() => handleSwipe("like", topUser)} />}

        <nav className="fixed bottom-0 left-1/2 grid w-full max-w-md -translate-x-1/2 grid-cols-5 border-t border-white/10 bg-zinc-950/95 px-3 pb-5 pt-3 text-[11px] backdrop-blur-xl">
          {["Discover", "Connect", "Events", "Map", "Profile"].map((item, i) => <div key={item} className={`text-center ${i === 0 ? "text-pink-400" : "text-zinc-500"}`}>{item}</div>)}
        </nav>
      </div>

      <AnimatePresence>{matchUser && <MatchModal user={matchUser} onClose={() => setMatchUser(null)} />}</AnimatePresence>

      <AnimatePresence>
        {preview && (
          <motion.div className="fixed inset-0 z-40 bg-black/80 p-5 backdrop-blur-xl" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={() => setPreview(null)}>
            <motion.div onClick={(e) => e.stopPropagation()} initial={{ y: 80 }} animate={{ y: 0 }} exit={{ y: 80, opacity: 0 }} className="mx-auto mt-10 max-w-md overflow-hidden rounded-[32px] border border-white/10 bg-zinc-950">
              {preview.photo ? (
                <img src={preview.photo} className="h-80 w-full object-cover" />
              ) : (
                <div className="grid h-80 w-full place-items-center bg-gradient-to-br from-zinc-900 via-fuchsia-950 to-black">
                  <div className="grid h-32 w-32 place-items-center rounded-full border border-white/15 bg-white/10 text-5xl font-black text-white">
                    {getInitials(preview.name)}
                  </div>
                </div>
              )}
              <div className="p-5">
                <h2 className="text-3xl font-black">{preview.name}, {preview.age}</h2>
                <p className="mt-2 text-zinc-300">{preview.bio}</p>
                <p className="mt-3 rounded-2xl bg-pink-500/15 p-3 text-sm text-pink-100">{preview.vibe}</p>
                <button onClick={() => setPreview(null)} className="mt-5 w-full rounded-full bg-gradient-to-r from-pink-500 to-purple-700 py-4 font-bold">Like</button>
              </div>
            </motion.div>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}
