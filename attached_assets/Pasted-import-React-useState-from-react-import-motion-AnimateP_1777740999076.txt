import React, { useState } from "react";
import { motion, AnimatePresence, useMotionValue, useTransform } from "framer-motion";
import {
  MapPin,
  ShieldCheck,
  Flame,
  Users,
  BriefcaseBusiness,
  Sparkles,
  ArrowLeft,
  ArrowRight,
  ArrowUp,
} from "lucide-react";

function SwipeDeck({ profiles, theme, activeIntent, onOpenProfile }) {
  const [index, setIndex] = useState(0);
  const [swipeAction, setSwipeAction] = useState(null);
  const [sparkBurst, setSparkBurst] = useState(false);

  const currentProfile = profiles[index % profiles.length];

  const completeSwipe = (action) => {
    setSwipeAction(action);

    if (action === "spark") {
      setSparkBurst(true);
      setTimeout(() => setSparkBurst(false), 650);
    }

    setTimeout(() => {
      setIndex((prev) => prev + 1);
      setSwipeAction(null);
    }, 360);
  };

  return (
    <div className="relative mt-4 h-[calc(100vh-330px)] min-h-[500px] max-h-[650px]">
      <GestureHints />

      <div className="absolute inset-x-4 bottom-[-16px] h-full rounded-[32px] border border-pink-500/15 bg-pink-500/5" />
      <div className="absolute inset-x-7 bottom-[-30px] h-full rounded-[32px] border border-pink-500/10 bg-pink-500/5" />

      <AnimatePresence mode="popLayout">
        <SwipeCard
          key={currentProfile.id}
          profile={currentProfile}
          theme={theme}
          action={swipeAction}
          onSwipe={completeSwipe}
          onOpenProfile={() => onOpenProfile(currentProfile)}
        />
      </AnimatePresence>

      <AnimatePresence>{sparkBurst && <SparkExplosion />}</AnimatePresence>
    </div>
  );
}

function SwipeCard({ profile, theme, action, onSwipe, onOpenProfile }) {
  const x = useMotionValue(0);
  const y = useMotionValue(0);

  const rotate = useTransform(x, [-220, 0, 220], [-12, 0, 12]);
  const scale = useTransform(y, [-180, 0], [1.04, 1]);

  const passOpacity = useTransform(x, [-160, -60, 0], [1, 0.45, 0]);
  const vibeOpacity = useTransform(x, [0, 60, 160], [0, 0.45, 1]);
  const sparkOpacity = useTransform(y, [-170, -70, 0], [1, 0.5, 0]);

  const handleDragEnd = (_, info) => {
    const offsetX = info.offset.x;
    const offsetY = info.offset.y;
    const velocityX = info.velocity.x;
    const velocityY = info.velocity.y;

    if (offsetY < -120 || velocityY < -900) {
      onSwipe("spark");
      return;
    }

    if (offsetX > 130 || velocityX > 900) {
      onSwipe("vibe");
      return;
    }

    if (offsetX < -130 || velocityX < -900) {
      onSwipe("pass");
    }
  };

  const exitAnimation =
    action === "pass"
      ? { x: -520, rotate: -22, opacity: 0 }
      : action === "vibe"
      ? { x: 520, rotate: 22, opacity: 0 }
      : action === "spark"
      ? { y: -620, scale: 1.1, opacity: 0 }
      : { opacity: 0 };

  return (
    <motion.article
      style={{ x, y, rotate, scale }}
      drag
      dragElastic={0.22}
      dragMomentum={false}
      onDragEnd={handleDragEnd}
      onClick={onOpenProfile}
      initial={{ y: 60, scale: 0.94, opacity: 0 }}
      animate={{ y: 0, scale: 1, opacity: 1 }}
      exit={exitAnimation}
      transition={{ type: "spring", stiffness: 260, damping: 28 }}
      className="absolute inset-0 cursor-grab overflow-hidden rounded-[32px] border border-pink-400/45 bg-black shadow-[0_0_55px_rgba(236,72,153,0.28)] active:cursor-grabbing"
    >
      <img
        src={profile.image}
        alt={profile.name}
        className="absolute inset-0 h-full w-full object-cover object-center"
      />

      <div className="absolute inset-0 bg-gradient-to-t from-black via-black/45 to-transparent" />

      <motion.div
        style={{ opacity: passOpacity }}
        className="absolute left-6 top-24 rotate-[-14deg] rounded-2xl border-2 border-rose-400 bg-rose-500/10 px-5 py-2 text-4xl font-black text-rose-400 shadow-[0_0_30px_rgba(244,63,94,0.5)]"
      >
        PASS
      </motion.div>

      <motion.div
        style={{ opacity: vibeOpacity }}
        className="absolute right-6 top-24 rotate-[14deg] rounded-2xl border-2 border-pink-400 bg-pink-500/10 px-5 py-2 text-4xl font-black text-pink-300 shadow-[0_0_30px_rgba(236,72,153,0.6)]"
      >
        VIBE
      </motion.div>

      <motion.div
        style={{ opacity: sparkOpacity }}
        className="absolute left-1/2 top-16 -translate-x-1/2 text-center"
      >
        <ArrowUp className="mx-auto h-10 w-10 text-emerald-300 drop-shadow-[0_0_18px_rgba(52,211,153,1)]" />
        <div className="mt-1 text-4xl font-black text-emerald-300 drop-shadow-[0_0_18px_rgba(52,211,153,1)]">
          SPARK
        </div>
        <p className="text-xs font-bold text-emerald-100">Stronger interest</p>
      </motion.div>

      <div className="absolute left-4 right-4 top-4 flex items-center justify-between">
        <div className="flex items-center gap-2 rounded-full border border-emerald-400/30 bg-black/55 px-3 py-1.5 text-xs font-black backdrop-blur-xl">
          <span className="h-2 w-2 rounded-full bg-emerald-400 shadow-[0_0_12px_rgba(52,211,153,1)]" />
          Online
        </div>

        <div className="grid h-[70px] w-[70px] place-items-center rounded-full border border-pink-400/50 bg-black/50 text-center backdrop-blur-xl shadow-[0_0_30px_rgba(236,72,153,0.45)]">
          <span className="text-xl font-black text-pink-100">{profile.matchScore}%</span>
          <span className="-mt-5 text-xs font-bold text-zinc-200">Match</span>
        </div>
      </div>

      <div className="absolute bottom-0 left-0 right-0 p-6">
        <div className="flex items-center gap-2">
          <h2 className="text-[38px] font-black leading-none text-white">
            {profile.name}, {profile.age}
          </h2>
          {profile.verified && (
            <ShieldCheck className="h-6 w-6 fill-pink-500 text-white" />
          )}
        </div>

        <div className="mt-2 flex items-center gap-2 text-sm text-zinc-200">
          <MapPin className="h-4 w-4" />
          <span>{profile.location}</span>
          <span className="h-1.5 w-1.5 rounded-full bg-emerald-400" />
          <span>{profile.distance}</span>
        </div>

        <div className="mt-3 flex flex-wrap gap-2">
          <span className={`rounded-full bg-gradient-to-r ${theme.gradient} px-4 py-1.5 text-xs font-black capitalize`}>
            {profile.intent}
          </span>
          <span className="rounded-full border border-white/15 bg-black/45 px-4 py-1.5 text-xs font-bold backdrop-blur-xl">
            {profile.subGenre}
          </span>
        </div>

        <p className="mt-4 text-sm font-semibold text-zinc-300">
          Tap card to view full profile
        </p>

        <div className="mt-6 rounded-3xl border border-white/10 bg-black/45 px-5 py-4 backdrop-blur-xl">
          <p className="text-center text-[11px] font-black tracking-[0.35em] text-zinc-300">
            SWIPE TO CONNECT
          </p>

          <div className="mt-4 grid grid-cols-3 text-center">
            <GestureMini icon={ArrowLeft} title="PASS" subtitle="Left" color="text-rose-400" />
            <GestureMini icon={ArrowUp} title="SPARK" subtitle="Up" color="text-emerald-300" />
            <GestureMini icon={ArrowRight} title="VIBE" subtitle="Right" color="text-pink-400" />
          </div>
        </div>
      </div>
    </motion.article>
  );
}

function GestureMini({ icon: Icon, title, subtitle, color }) {
  return (
    <div className="flex flex-col items-center">
      <Icon className={`h-6 w-6 ${color} drop-shadow-[0_0_12px_currentColor]`} />
      <p className={`mt-2 text-xs font-black ${color}`}>{title}</p>
      <p className="text-xs text-zinc-400">{subtitle}</p>
    </div>
  );
}

function SparkExplosion() {
  const particles = Array.from({ length: 22 });

  return (
    <motion.div
      className="pointer-events-none absolute inset-0 z-50 flex items-center justify-center"
      initial={{ opacity: 1 }}
      exit={{ opacity: 0 }}
    >
      {particles.map((_, i) => {
        const angle = (i / particles.length) * Math.PI * 2;
        const distance = 90 + Math.random() * 90;
        const x = Math.cos(angle) * distance;
        const y = Math.sin(angle) * distance;

        return (
          <motion.span
            key={i}
            initial={{ x: 0, y: 0, scale: 0, opacity: 1 }}
            animate={{ x, y, scale: [0, 1.2, 0], opacity: [1, 1, 0] }}
            transition={{ duration: 0.65, ease: "easeOut" }}
            className="absolute h-2 w-2 rounded-full bg-emerald-300 shadow-[0_0_16px_rgba(52,211,153,1)]"
          />
        );
      })}

      <motion.div
        initial={{ scale: 0, opacity: 1 }}
        animate={{ scale: 1.6, opacity: 0 }}
        transition={{ duration: 0.55 }}
        className="absolute grid h-28 w-28 place-items-center rounded-full border border-emerald-300/70 bg-emerald-400/10 shadow-[0_0_50px_rgba(52,211,153,0.8)]"
      >
        <Sparkles className="h-12 w-12 text-emerald-300" />
      </motion.div>
    </motion.div>
  );
