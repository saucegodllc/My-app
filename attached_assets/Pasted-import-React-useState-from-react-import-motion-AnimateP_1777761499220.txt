export default function DoubleDateFeature() {
  const [tab, setTab] = useState("discover");
  const [index, setIndex] = useState(0);
  const [showModal, setShowModal] = useState(false);

  const profile = duoProfiles[index % duoProfiles.length];

  const nextCard = () => setIndex((prev) => prev + 1);

  return (
    <div className="min-h-screen bg-gradient-to-b from-[#120816] via-[#1d0d26] to-black text-white px-4 pt-12 pb-6 overflow-hidden">
      <div className="max-w-md mx-auto">

              <Users size={21} />
            </div>
            <div>
              <h1 className="text-3xl font-black tracking-tight">Double Date</h1>
              <p className="text-white/60 text-sm">Match as a duo on ConnectSphere.</p>
            </div>
          </div>
        </div>

        <div className="grid grid-cols-3 gap-2 bg-white/8 p-1 rounded-2xl mb-5 backdrop-blur">
          {[
            ["discover", "Discover"],
            ["myduo", "My Duo"],
            ["requests", "Requests"],
          ].map(([key, label]) => (
            <button
              key={key}
              onClick={() => setTab(key)}
              className={`py-3 rounded-xl text-sm font-bold transition ${
                tab === key
                  ? "bg-white text-black shadow-lg"
                  : "text-white/60"
              }`}
            >
              {label}
            </button>
          ))}
        </div>

        {tab === "discover" && (
          <DuoProfileCard
            profile={profile}
            onPass={nextCard}
            onSpark={() => setShowModal(true)}
            onInvite={() => setShowModal(true)}
          />
        )}

        {tab === "myduo" && <MyDuoSetup />}

        {tab === "requests" && <DuoRequests />}

        <AnimatePresence>
          {showModal && <MatchModal onClose={() => setShowModal(false)} />}
        </AnimatePresence>
      </div>
    </div>
  );
}

function ActionButton({ icon, label, onClick, special }) {
  return (
    <motion.button
      whileTap={{ scale: 0.88, rotate: special ? 4 : 0 }}
      whileHover={{ scale: 1.04 }}
      onClick={onClick}
      className={`h-16 rounded-2xl flex flex-col items-center justify-center gap-1 font-black shadow-xl ${
        special
          ? "bg-gradient-to-br from-purple-400 to-pink-500 text-white"
          : "bg-white text-black"
      }`}
    >
      <div className="scale-110">{icon}</div>
      <span className="text-xs">{label}</span>
    </motion.button>
  );
}
